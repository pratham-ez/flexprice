## ingest

Ingest a new event into the system

### Example Usage

<!-- UsageSnippet language="typescript" operationID="post_/events" method="post" path="/events" -->
```typescript
import { Flexprice } from "@flexprice/sdk";

const flexprice = new Flexprice({
  serverURL: "https://api.example.com",
  apiKeyAuth: "<YOUR_API_KEY_HERE>",
});

async function run() {
  const result = await flexprice.events.ingest({
    customerId: "customer456",
    eventId: "event123",
    eventName: "api_request",
    externalCustomerId: "customer456",
    properties: {
      ""response_status"": "200}",
      "{"request_size"": "100",
    },
    source: "api",
    timestamp: "2024-03-20T15:04:05Z",
  });

  console.log(result);
}

run();
```
